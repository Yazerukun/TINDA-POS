-- 1. Owner Accounts
CREATE TABLE IF NOT EXISTS public.owner_users (
  id BIGSERIAL PRIMARY KEY,
  username TEXT NOT NULL UNIQUE,
  password_hash TEXT NOT NULL,
  password_salt TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 2. Store Branches (One per physical store terminal / PC)
CREATE TABLE IF NOT EXISTS public.stores (
  id TEXT PRIMARY KEY,
  branch_name TEXT NOT NULL,
  sync_key_hash TEXT NOT NULL,
  owner_user_id BIGINT REFERENCES public.owner_users(id) ON DELETE SET NULL,
  last_sale_at TIMESTAMPTZ,
  last_sync_at TIMESTAMPTZ,
  pin_hash TEXT,
  slug TEXT,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 2b. Store Owners (Many-to-Many Store Ownership Mapping)
CREATE TABLE IF NOT EXISTS public.store_owners (
  store_id TEXT NOT NULL REFERENCES public.stores(id) ON DELETE CASCADE,
  owner_user_id BIGINT NOT NULL REFERENCES public.owner_users(id) ON DELETE CASCADE,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (store_id, owner_user_id)
);

-- 3. Sales Header Table
CREATE TABLE IF NOT EXISTS public.sales (
  id BIGSERIAL PRIMARY KEY,
  store_id TEXT NOT NULL REFERENCES public.stores(id) ON DELETE CASCADE,
  local_sale_id BIGINT NOT NULL,
  transaction_no TEXT NOT NULL,
  cashier_name TEXT,
  customer_name TEXT,
  subtotal_c BIGINT NOT NULL DEFAULT 0,
  discount_c BIGINT NOT NULL DEFAULT 0,
  total_c BIGINT NOT NULL DEFAULT 0,
  status TEXT NOT NULL DEFAULT 'COMPLETED',
  cash_c BIGINT NOT NULL DEFAULT 0,
  gcash_c BIGINT NOT NULL DEFAULT 0,
  maya_c BIGINT NOT NULL DEFAULT 0,
  utang_c BIGINT NOT NULL DEFAULT 0,
  local_shift_id BIGINT,
  sold_at TIMESTAMPTZ NOT NULL,
  synced_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT uq_sales_store_local UNIQUE (store_id, local_sale_id)
);

CREATE INDEX IF NOT EXISTS idx_sales_store ON public.sales(store_id);
CREATE INDEX IF NOT EXISTS idx_sales_date ON public.sales(sold_at);
CREATE INDEX IF NOT EXISTS idx_sales_store_date ON public.sales(store_id, sold_at);

-- 4. Sale Items (Line Items)
CREATE TABLE IF NOT EXISTS public.sale_items (
  id BIGSERIAL PRIMARY KEY,
  sale_id BIGINT NOT NULL REFERENCES public.sales(id) ON DELETE CASCADE,
  product_name TEXT NOT NULL,
  unit_name TEXT NOT NULL,
  qty NUMERIC NOT NULL DEFAULT 1,
  unit_price_c BIGINT NOT NULL,
  subtotal_c BIGINT NOT NULL
);

CREATE INDEX IF NOT EXISTS idx_sale_items_sale ON public.sale_items(sale_id);

-- 5. Stock Movements
CREATE TABLE IF NOT EXISTS public.stock_movements (
  id BIGSERIAL PRIMARY KEY,
  store_id TEXT NOT NULL REFERENCES public.stores(id) ON DELETE CASCADE,
  local_movement_id BIGINT NOT NULL,
  product_name TEXT NOT NULL,
  movement_type TEXT NOT NULL,
  quantity_change NUMERIC NOT NULL,
  quantity_before NUMERIC NOT NULL,
  quantity_after NUMERIC NOT NULL,
  unit TEXT NOT NULL DEFAULT 'piece',
  reference TEXT,
  reason TEXT,
  moved_at TIMESTAMPTZ NOT NULL,
  synced_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT uq_movements_store_local UNIQUE (store_id, local_movement_id)
);

CREATE INDEX IF NOT EXISTS idx_movements_store ON public.stock_movements(store_id);
CREATE INDEX IF NOT EXISTS idx_movements_date ON public.stock_movements(moved_at);
CREATE INDEX IF NOT EXISTS idx_movements_product ON public.stock_movements(store_id, product_name);

-- 6. Shift Snapshots (Shift Reports)
CREATE TABLE IF NOT EXISTS public.shift_snapshots (
  id BIGSERIAL PRIMARY KEY,
  store_id TEXT NOT NULL REFERENCES public.stores(id) ON DELETE CASCADE,
  local_shift_id BIGINT NOT NULL,
  cashier_name TEXT NOT NULL,
  shift_date TEXT NOT NULL,
  opened_at TIMESTAMPTZ NOT NULL,
  closed_at TIMESTAMPTZ,
  gross_sales_c BIGINT NOT NULL DEFAULT 0,
  cash_c BIGINT NOT NULL DEFAULT 0,
  gcash_c BIGINT NOT NULL DEFAULT 0,
  maya_c BIGINT NOT NULL DEFAULT 0,
  utang_c BIGINT NOT NULL DEFAULT 0,
  refund_total_c BIGINT NOT NULL DEFAULT 0,
  void_count INT NOT NULL DEFAULT 0,
  transaction_count INT NOT NULL DEFAULT 0,
  expenses_c BIGINT NOT NULL DEFAULT 0,
  net_sales_c BIGINT NOT NULL DEFAULT 0,
  synced_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CONSTRAINT uq_shifts_store_local UNIQUE (store_id, local_shift_id)
);

CREATE INDEX IF NOT EXISTS idx_shifts_store ON public.shift_snapshots(store_id);
CREATE INDEX IF NOT EXISTS idx_shifts_date ON public.shift_snapshots(shift_date);

-- 7. Owner Sessions
CREATE TABLE IF NOT EXISTS public.owner_sessions (
  id BIGSERIAL PRIMARY KEY,
  owner_user_id BIGINT NOT NULL REFERENCES public.owner_users(id) ON DELETE CASCADE,
  token_hash TEXT NOT NULL UNIQUE,
  expires_at TIMESTAMPTZ NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_sessions_token ON public.owner_sessions(token_hash);

-- 8. Global Community Chat Messages
CREATE TABLE IF NOT EXISTS public.global_chat_messages (
  id BIGSERIAL PRIMARY KEY,
  store_id TEXT NOT NULL REFERENCES public.stores(id) ON DELETE CASCADE,
  branch_name TEXT NOT NULL,
  sender_name TEXT NOT NULL,
  message TEXT NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 9. Live Announcements
CREATE TABLE IF NOT EXISTS public.live_announcements (
  id BIGSERIAL PRIMARY KEY,
  title TEXT NOT NULL,
  message TEXT NOT NULL,
  severity TEXT NOT NULL DEFAULT 'info',
  active INT NOT NULL DEFAULT 1,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 10. Real-Time Telemetry & Error Radar
CREATE TABLE IF NOT EXISTS public.system_telemetry (
  id BIGSERIAL PRIMARY KEY,
  store_id TEXT NOT NULL REFERENCES public.stores(id) ON DELETE CASCADE,
  health_score INT NOT NULL,
  error_count INT NOT NULL DEFAULT 0,
  flight_recorder JSONB,
  reported_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

CREATE INDEX IF NOT EXISTS idx_telemetry_store ON public.system_telemetry(store_id);
